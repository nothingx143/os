# Question 93: Hashing - T-Rex And The Pairs (Array Difference Equation)

## Verified Solution (`Q093.c`)

```c
#include <stdio.h>
#include <stdlib.h>

#define MAXN 100005

struct Node {
    long long val;
    int idx;
} arr[MAXN];

long long a[MAXN];
int n;

int cmfn1(const void *a,const void *b) {
    const struct Node *x = (const struct Node*)a;
    const struct Node *y = (const struct Node*)b;
    if (x->val < y->val) return -1;
    if (x->val > y->val) return 1;
    return x->idx - y->idx;
}

#define LESS(mid) (arr[mid].val < tv || (arr[mid].val == tv && arr[mid].idx < ti))

int lower_bound(long long tv, int ti) {
    int l = 0, r = n, mid;
    while (l < r) {
        mid = (l + r) / 2;
        if (LESS(mid)) l = mid + 1;
        else r = mid;
    }
    return l;
}

int main() {
    scanf("%d", &n);
    for (int i = 1; i <= n; i++) {
        scanf("%lld", &a[i]);
        arr[i - 1].val = a[i] + 1LL * i * i;
        arr[i - 1].idx = i;
    }
    qsort(arr, n, sizeof(struct Node), cmfn1);
    long long ans = 0;
    for (int j = 1; j <= n; j++) {
        long long target = a[j] - 1LL * j * j;
        int left = lower_bound(target, 0);
        int right = lower_bound(target, j);
        ans += (right - left);
    }
    printf("%lld\n", ans);
    return 0;
}
```

## Problem Description
Given an array $A$ of length $N$ (1-based), count the number of pairs $(i, j)$ with $1 \le i < j \le N$ such that:
$$A[j] - A[i] = i^2 + j^2$$

## Constraints
- $1 \le N \le 10^5$
- $1 \le A[i] \le 10^{12}$

## Solution Approach
- Rearrange the equation to isolate index variables:
  $$A[j] - j^2 = A[i] + i^2$$
- Define $B[i] = A[i] + i^2$. Store each $(B[i], i)$ in an array and sort by value, then index, using `qsort` with comparator `cmfn1`.
- For each $j \in [1, N]$, the target value is $T = A[j] - j^2$. Use binary search (`lower_bound`) to count the number of elements with value equal to $T$ and index $i < j$.
- Runs in $O(N \log N)$ time and $O(N)$ space, strictly adhering to the static complexity constraints ($\text{CCN} \le 3$).
