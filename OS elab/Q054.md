# Q054: Queue - Club on Eerie Planet (Level 1 Challenge 54)

## Verified Solution (`Q054.c`)

```c
#include <stdio.h>
#include <stdlib.h>

struct Crew {
    long long h;
    long long start;
    long long end;
} crew[100005];

int crew_cnt = 0;

void enqueue(long long h,long long start,long long end) {
    crew[crew_cnt].h = h;
    crew[crew_cnt].start = start;
    crew[crew_cnt].end = end;
    crew_cnt++;
}

int main() {
    long long H;
    int c, q;
    if (scanf("%lld%d%d", &H, &c, &q) != 3) return 0;
    while(c--) {
        long long h, s, e;
        if (scanf("%lld%lld%lld", &h, &s, &e) == 3) {
            enqueue(h, s, e);
        }
    }
    for (int k = 0; k < q; k++) {
        long long qh = 0, qt = 0;
        if (scanf("%lld%lld", &qh, &qt) != 2) {
            puts("YES");
            continue;
        }
        int allowed = 1;
        for (int i = 0; i < crew_cnt; i++) {
            if (qt >= crew[i].start && qt <= crew[i].end) {
                if (qh <= crew[i].h) {
                    allowed = 0;
                    break;
                }
            }
        }
        if (allowed) puts("YES");
        else puts("NO");
    }
    return 0;
}
```

## Problem Statement

You own a club on an eerie planet. The day on this planet comprises of $H$ hours. You appointed $C$ crew members to handle the huge crowd. Each crew member has a height $h_i$ and works a shift from hour $s_i$ to $e_i$. A guest arriving at hour $qt$ with height $qh$ is allowed entry only if they are strictly taller than every crew member currently on duty at that hour. Answer $Q$ queries determining whether each visitor is allowed entry (`YES` or `NO`).

### Input Format
- Line 1: Three integers $H$ (hours in day), $C$ (number of crew members), and $Q$ (number of queries).
- Next $C$ lines: Three integers $h$, $s$, $e$ representing each crew member's height and duty interval $[s, e]$.
- Next $Q$ lines: Two integers $qh$ and $qt$ representing the guest's height and arrival time.

### Output Format
- For each query, print `YES` if the guest is allowed entry, or `NO` otherwise.
